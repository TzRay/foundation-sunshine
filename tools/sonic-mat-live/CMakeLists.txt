cmake_minimum_required(VERSION 3.25)
project(mat_capture LANGUAGES C CXX)
if (NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MAT capture tests require x64 MSVC and the Windows SDK")
endif ()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8)
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
option(MAT_BUILD_TOOLS "Build the opt-in native capture tools" ON)
include(CTest)
include(FetchContent)
FetchContent_Declare(minhook
        GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
        GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
get_filename_component(root "../.." ABSOLUTE)

add_library(sunshine-sonic-mat-worker SHARED "${root}/src/spatial_audio/mat_stream_receiver.cpp" native_output_tap.cpp)
target_link_libraries(sunshine-sonic-mat-worker PRIVATE runtimeobject ole32 bcrypt minhook)
if (MAT_BUILD_TOOLS)
    add_library(sunshine-plugin-object-tap SHARED plugin_tap.cpp "${root}/tools/mat-capture/quantum_v4_recorder.cpp")
    target_compile_definitions(sunshine-plugin-object-tap PRIVATE SPATIAL_USE_EXTERNAL_MINHOOK)
    target_link_libraries(sunshine-plugin-object-tap PRIVATE minhook)
    add_executable(sunshine-plugin-mat-capture plugin_capture.cpp bed_capture.cpp bed_manager.cpp native_output_capture.cpp audio_sessions.cpp)
    target_link_libraries(sunshine-plugin-mat-capture PRIVATE bcrypt version ole32 uuid propsys shell32 taskschd comsuppw avrt)
    add_executable(spatial-probe spatial_probe.cpp)
    target_link_libraries(spatial-probe PRIVATE ole32 uuid propsys)
    add_executable(bed-probe bed_probe.cpp)
    target_link_libraries(bed-probe PRIVATE bcrypt version)
    add_executable(bed-source-probe bed_source_probe.cpp)
    target_link_libraries(bed-source-probe PRIVATE ole32 uuid)
    add_library(sunshine-native-output-tap SHARED native_output_tap.cpp)
    # R16's pinned diagnostic DLL can remain loaded; never reuse its basename.
    set_target_properties(sunshine-native-output-tap PROPERTIES OUTPUT_NAME sunshine-native-endpoint-tap)
    target_link_libraries(sunshine-native-output-tap PRIVATE minhook bcrypt)
    add_executable(native-output-capture native_output_capture.cpp)
    target_compile_definitions(native-output-capture PRIVATE MAT_OUTPUT_STANDALONE)
    target_link_libraries(native-output-capture PRIVATE bcrypt version ole32 uuid avrt)
endif ()

    set_tests_properties(mat_manager PROPERTIES TIMEOUT 10)

    add_executable(mat_bed_route_tests bed_route_test.cpp)
    add_executable(mat_compat_tests compat_test.cpp)
    target_link_libraries(mat_compat_tests PRIVATE bcrypt version)
    add_executable(mat_queue_tests queue_test.cpp)
    add_dependencies(mat_queue_tests sunshine-sonic-mat-worker)
    add_executable(mat_callback_tests
            "${root}/tools/mat-capture/quantum_v4_selftest.cpp"

    set_tests_properties(mat_capture_options mat_bridge mat_bed_input mat_bed_route mat_compat mat_callbacks mat_dynamic_ipc mat_bed714_ipc mat_source_clock_ipc mat_pipe mat_native_output mat_native_relay
            PROPERTIES TIMEOUT 30)
